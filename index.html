Pour que le calcul des points fonctionne, tu dois copier ces règles dans Firebase Console → Firestore → Règles :

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true;
    }
    function isAuth() { return request.auth != null; }

    match /users/{userId} {
      allow read: if isAuth();
      allow create: if request.auth.uid == userId;
      allow update: if request.auth.uid == userId || isAdmin();
    }
    match /matches/{matchId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }
    match /predictions/{predId} {
      allow read: if isAuth();
      allow create: if isAuth() && request.auth.uid == request.resource.data.uid;
      allow update: if isAuth() && (resource.data.uid == request.auth.uid || isAdmin());
    }
    match /settings/{docId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }
  }
}

⚡ La ligne clé : allow update: if isAuth() && (resource.data.uid == request.auth.uid || isAdmin()) — elle autorise l'admin à mettre à jour les points.
